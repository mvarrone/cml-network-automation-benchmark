$round = 1
$intervalSeconds = 1

while ($true) {
    Clear-Host

    Write-Host "Ping sweep - Round $round - $(Get-Date)"
    Write-Host ""

    $results = 11..20 | ForEach-Object {
        $ip = "192.168.1.$_"
        $ok = Test-Connection $ip -Count 1 -Quiet

        [PSCustomObject]@{
            IP     = $ip
            Status = if ($ok) { "UP" } else { "DOWN" }
        }
    }

    $results | Format-Table -AutoSize | Out-String | Write-Host

    $upCount = ($results | Where-Object Status -eq "UP").Count

    Write-Host "Devices UP: $upCount/10"
    Write-Host ""

    $timeUnit = if ($intervalSeconds -eq 1) { "second" } else { "seconds" }

    Write-Host "Next round in $intervalSeconds $timeUnit... (Ctrl+C to stop)"

    $round++
    Start-Sleep -Seconds $intervalSeconds
}
